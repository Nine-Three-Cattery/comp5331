\begin{frame}{A graph-conditioned distribution of future paths}
\begin{enumerate}
\item Target and 3 m pedestrian-edge histories feed two recurrent encoders.
\item The history-conditioned prior samples one of 25 latent categories.
\item A GRU samples correlated velocities and integrates them in meters.
\end{enumerate}
\begin{block}{Training and inference}
The training posterior sees future labels. Test sampling uses observed
target history and observed neighbor context.
\end{block}
\end{frame}

\begin{frame}{The benchmark enforces what is visible at the anchor}
\begin{itemize}
\item Five scene holdouts, three training seeds, $\Delta t=0.4$ s.
\item Choose checkpoints on clean validation Energy Score with $K=20$.
\item Centered derivatives leaked future coordinates into old inputs.
\item Recompute motion with window-local backward differences for targets and neighbors.
\end{itemize}
\begin{alertblock}{Regression criterion}
Change a target or neighbor after the anchor. Features and same-seed
predictions must remain identical in all six conditions.
\end{alertblock}
\end{frame}
